using System.Reflection;
using System.Text.Json;
using LealInfoPDV;

internal static class Program
{
    static IEnumerable<Control> Children(Control parent)
    {
        foreach (Control child in parent.Controls)
        {
            yield return child;
            foreach (var descendant in Children(child)) yield return descendant;
        }
    }
    [STAThread]
    static void Main()
    {
        Application.SetHighDpiMode(HighDpiMode.PerMonitorV2);
        Application.EnableVisualStyles();
        Application.SetCompatibleTextRenderingDefault(false);
        Application.SetUnhandledExceptionMode(UnhandledExceptionMode.ThrowException);
        Directory.CreateDirectory("visual-qa");
        AppDomain.CurrentDomain.UnhandledException += (_, e) => File.WriteAllText("visual-qa/error.txt", e.ExceptionObject.ToString());
        WeatherChecks.RunAsync().GetAwaiter().GetResult();
        SalesWeatherCard.ServiceFactory = () => WeatherChecks.FixtureService();
        Database.Initialize();
        using (var cn = Database.Open())
        using (var cmd = cn.CreateCommand())
        {
            cmd.CommandText = "INSERT INTO products(barcode,name,price,stock,active) VALUES('QA-THEME','Produto QA',10,100,1); INSERT OR REPLACE INTO settings(key,value) VALUES('pix_key','qa@example.com'),('pix_key_type','EMAIL'),('pix_mp_enabled','0'),('company_city_state','Angra dos Reis/RJ');";
            cmd.ExecuteNonQuery();
        }
        using var owner = new MainForm();
        using var timer = new System.Windows.Forms.Timer { Interval = 300 };
        var themes = new[] { "Azul Texturizado", "Futurista Azul", "Clean Pro", "Dark Premium", "Blue Red Racing", "PDV Rosa" };
        int themeIndex = 0;
        int stage = 0;
        bool reopening = false;
        var checks = new List<string>();
        void Capture(Form sales, string name)
        {
            bool responsive = sales.Tag is SalesResponsiveLayout;
            using var bitmap = new Bitmap(responsive ? sales.ClientSize.Width : sales.Width, responsive ? sales.ClientSize.Height : sales.Height);
            if (responsive)
            {
                // CI's virtual monitor is 1024px. Render the real root controls
                // directly so the simulated larger viewport is not clipped by it.
                using (var graphics = Graphics.FromImage(bitmap)) graphics.Clear(sales.BackColor);
                foreach (Control root in sales.Controls)
                    if (root.Visible && root.Width > 0 && root.Height > 0)
                        root.DrawToBitmap(bitmap, root.Bounds);
            }
            else sales.DrawToBitmap(bitmap, new Rectangle(0, 0, bitmap.Width, bitmap.Height));
            bitmap.Save("visual-qa/" + name + ".png", System.Drawing.Imaging.ImageFormat.Png);
        }
        void Verify(Form sales, string theme)
        {
            var palette = SalesPalette.For(theme);
            var weather = Children(sales).OfType<SalesWeatherCard>().Single();
            if (weather.Palette != palette) throw new Exception("Tema do tempo divergente");
            if (weather.Reading?.Location.City != "Angra dos Reis") throw new Exception("Cidade do painel divergente");
            var controls = Children(sales).ToArray();
            var grid = controls.OfType<SalesItemsGrid>().Single();
            var status = controls.OfType<SalesGlowLabel>().Single();
            if (sales.BackColor != palette.Background || grid.BackgroundColor != palette.Background || grid.DefaultCellStyle.ForeColor != palette.Foreground || status.ForeColor != palette.Foreground || status.GlowColor != palette.Accent)
                throw new Exception("Cores nao sincronizadas: " + theme);
            bool textured = theme is "PDV Rosa" or "Azul Texturizado";
            if ((sales.BackgroundImage != null) != textured || (grid.ThemeTexture != null) != textured)
                throw new Exception("Textura original ausente ou vazando de outro tema: " + theme);
            if (textured)
            {
                using var source = new Bitmap(sales.BackgroundImage!);
                using var tableTexture = new Bitmap(grid.ThemeTexture!);
                if (source.Size != tableTexture.Size || source.GetPixel(source.Width / 2, source.Height / 2) != tableTexture.GetPixel(tableTexture.Width / 2, tableTexture.Height / 2))
                    throw new Exception("Textura original divergente: " + theme);
            }
            checks.Add((reopening ? "Reabertura: " : "Selecao real: ") + theme + " OK");
            File.WriteAllText("visual-qa/theme-checks.json", JsonSerializer.Serialize(checks));
        }
        void SendKey(Control control, Keys key) => typeof(Control).GetMethod("OnKeyDown", BindingFlags.Instance | BindingFlags.NonPublic)!.Invoke(control, new object[] { new KeyEventArgs(key) });
        Form Dialog(string title) => Application.OpenForms.Cast<Form>().Single(form => form.Text == title);
        void VerifyDialog(Form dialog, string name)
        {
            var palette = SalesPalette.For(themes[themeIndex]);
            bool textured = themes[themeIndex] is "PDV Rosa" or "Azul Texturizado";
            if (dialog.BackColor != palette.Background || dialog.ForeColor != palette.Foreground || (dialog.BackgroundImage != null) != textured)
                throw new Exception("Tema da janela divergente: " + name);
            foreach (var control in Children(dialog))
            {
                if (control is TextBoxBase or ComboBox or NumericUpDown)
                    if (control.BackColor != palette.Field || control.ForeColor != palette.FieldText) throw new Exception("Campo divergente: " + name);
                if (control is DataGridView grid)
                    if (grid.BackgroundColor != palette.Background || grid.DefaultCellStyle.ForeColor != palette.Foreground) throw new Exception("Tabela divergente: " + name);
                if (control is TabPage page)
                    if (page.BackColor != palette.Background || page.UseVisualStyleBackColor) throw new Exception("Aba divergente: " + name);
            }
            checks.Add(name + ": " + themes[themeIndex] + " OK");
            File.WriteAllText("visual-qa/theme-checks.json", JsonSerializer.Serialize(checks));
            Capture(dialog, name + "-" + themeIndex);
        }
        async Task VerifyResponsiveMatrix(Form sales)
        {
            var responsive = (SalesResponsiveLayout)sales.Tag!;
            var report = new List<object>();
            foreach (var resolution in new[] { new Size(1366, 768), new Size(1600, 900), new Size(1920, 1080) })
            foreach (int dpi in new[] { 96, 120, 144 })
            {
                // Exercise the production calculation with physical client space:
                // taskbar and non-client chrome grow with Windows scaling.
                sales.ClientSize = new Size(resolution.Width - (int)(16 * dpi / 96f), resolution.Height - (int)(79 * dpi / 96f));
                responsive.Apply(dpi);
                sales.PerformLayout();
                var controls = Children(sales).ToArray();
                var status = controls.OfType<SalesGlowLabel>().Single();
                var statusCenter = sales.PointToClient(status.PointToScreen(new Point(status.Width / 2, 0))).X;
                if (Math.Abs(statusCenter - sales.ClientSize.Width / 2) > 3) throw new Exception("Status fora do centro");
                foreach (var button in controls.OfType<SalesVisualButton>().Where(c => c.Visible))
                    if (button.Width < 32 || button.Height < 28) throw new Exception("Botao espremido: " + button.Text);
                foreach (var c in controls.Where(c => c.Visible && c.Height > 0 && c.Width > 0 && c.Parent != null))
                {
                    if (c.Parent is NumericUpDown) continue;
                    if (c.Bounds.Left < 0 || c.Bounds.Top < 0 || c.Bounds.Right > c.Parent!.ClientSize.Width + 1 || c.Bounds.Bottom > c.Parent.ClientSize.Height + 1)
                        throw new Exception($"Controle cortado: {c.GetType().Name} {c.Text} {c.Bounds} em {c.Parent.ClientSize}");
                }
                var weather = controls.OfType<SalesWeatherCard>().Single();
                if (weather.Height < 130 || weather.Width < 200) throw new Exception("Clima espremido");
                var photo = controls.OfType<PictureBox>().Single(c => c.Parent != null && c.Parent.Parent is TableLayoutPanel);
                if (photo.Height < 170 || photo.SizeMode != PictureBoxSizeMode.Zoom) throw new Exception("Foto cortada");
                var before = controls.Select(c => c.Bounds).ToArray();
                using (var cn = Database.Open()) using (var cmd = cn.CreateCommand())
                { cmd.CommandText = "UPDATE settings SET value='' WHERE key='company_city_state'"; cmd.ExecuteNonQuery(); }
                await weather.RefreshWeatherAsync();
                if (weather.Reading != null || !before.SequenceEqual(controls.Select(c => c.Bounds))) throw new Exception("Falha do clima alterou layout");
                Capture(sales, $"responsive-{resolution.Width}x{resolution.Height}-{dpi}");
                responsive.Apply(dpi);
                if (!before.SequenceEqual(controls.Select(c => c.Bounds))) throw new Exception("Resize acumulativo: " + string.Join("; ", controls.Select((c, i) => (c, i)).Where(x => before[x.i] != x.c.Bounds).Select(x => $"{x.c.GetType().Name} {x.c.Text}: {before[x.i]} -> {x.c.Bounds}")));
                using (var cn = Database.Open()) using (var cmd = cn.CreateCommand())
                { cmd.CommandText = "UPDATE settings SET value='Angra dos Reis/RJ' WHERE key='company_city_state'"; cmd.ExecuteNonQuery(); }
                await weather.RefreshWeatherAsync();
                report.Add(new { Resolution = resolution.ToString(), Dpi = dpi, ActualDeviceDpi = sales.DeviceDpi, Client = sales.ClientSize.ToString(), StatusCenter = statusCenter, Weather = weather.Bounds.ToString(), Result = "OK" });
            }
            File.WriteAllText("visual-qa/responsive-matrix.json", JsonSerializer.Serialize(report));
            sales.Bounds = new Rectangle(0, 0, 1366, 730);
            responsive.Apply();
        }
        timer.Tick += async (_, _) =>
        {
            var sales = Application.OpenForms.Cast<Form>().FirstOrDefault(f => f.Text.Contains("TELA DE VENDA"));
            if (sales == null) return;
            if (stage == 0)
            {
                sales.MaximumSize = new Size(4000, 2400);
                sales.WindowState = FormWindowState.Normal;
                sales.Bounds = new Rectangle(0, 0, 1366, 730);
                sales.PerformLayout();
                stage = 1;
                return;
            }
            if (reopening)
            {
                Verify(sales, themes[^1]);
                Capture(sales, "sales-reopened");
                timer.Stop(); sales.Close(); return;
            }
            if (stage == 1)
            {
                if (themeIndex == 0)
                {
                    stage = -1;
                    await VerifyResponsiveMatrix(sales);
                    Capture(sales, "sales-1366");
                }
                stage = 2;
                Children(sales).OfType<Button>().Single(b => b.Text == "ESTILO").PerformClick();
                return;
            }
            if (stage == 2)
            {
                var chooser = Application.OpenForms.Cast<Form>().Single(f => f.Text == "Estilo da Tela de Vendas");
                stage = 3;
                Children(chooser).OfType<Button>().Single(b => b.Tag?.ToString() == themes[themeIndex]).PerformClick();
                return;
            }
            if (stage == 3)
            {
                Verify(sales, themes[themeIndex]); Capture(sales, "theme-" + themeIndex);
                stage = 4; Children(sales).OfType<SalesVisualButton>().Single(button => button.Text == "F5").PerformClick(); return;
            }
            if (stage == 4)
            {
                var catalog = Dialog("Consultar Produto - F5"); VerifyDialog(catalog, "F5-consulta");
                var grid = Children(catalog).OfType<DataGridView>().Single(); grid.CurrentCell = grid.Rows[0].Cells[0];
                stage = 5; Children(catalog).OfType<Button>().Single(button => button.Text == "SELECIONAR").PerformClick(); return;
            }
            if (stage == 5)
            {
                var quantity = Dialog("Quantidade do Produto"); VerifyDialog(quantity, "quantidade");
                stage = 6; quantity.DialogResult = DialogResult.Cancel; quantity.Close(); return;
            }
            if (stage == 6) { stage = 7; SendKey(sales, Keys.F10); return; }
            if (stage == 7)
            {
                var loose = Dialog("Venda Avulsa - F10"); VerifyDialog(loose, "F10-avulsa");
                var inputs = Children(loose).OfType<TextBox>().OrderBy(box => box.Top).ToArray();
                if (inputs[0].Text != "VENDA AVULSA" || !inputs[0].ReadOnly) throw new Exception("Descricao padrao incorreta");
                if (themeIndex % 2 == 1)
                {
                    Children(loose).OfType<Button>().Single(button => button.Text == "+").PerformClick();
                    if (inputs[0].ReadOnly || inputs[0].Text != "") throw new Exception("Botao + nao liberou a descricao");
                    inputs[0].Text = "Servico QA";
                    Capture(loose, "F10-personalizada-" + themeIndex);
                }
                inputs[1].Text = "1000";
                if (inputs[1].Text != "10,00") throw new Exception("Valor da venda avulsa alterado");
                stage = 8; Children(loose).OfType<Button>().Single(button => button.Text == "ADICIONAR À VENDA").PerformClick(); return;
            }
            if (stage == 8)
            {
                var grid = Children(sales).OfType<SalesItemsGrid>().Single();
                string expected = themeIndex % 2 == 0 ? "VENDA AVULSA" : "Servico QA";
                if (grid.Rows[0].Cells[1].Value?.ToString() != expected) throw new Exception("Descricao incorreta no carrinho");
                stage = 9; SendKey(sales, Keys.F4); return;
            }
            if (stage == 9)
            {
                var payment = Dialog("Finalizar Venda"); VerifyDialog(payment, "F4-pagamento");
                var tabs = Children(payment).OfType<TabControl>().Single();
                for (int index = 0; index < tabs.TabCount; index++) { tabs.SelectedIndex = index; Capture(payment, "pagamento-" + themeIndex + "-" + index); }
                tabs.SelectedIndex = 1; stage = 10;
                Children(payment).OfType<Button>().Single(button => button.Text.StartsWith("GERAR QR CODE PIX")).PerformClick(); return;
            }
            if (stage == 10)
            {
                var pix = Dialog("LEAL INFO PDV • PIX POR CHAVE"); VerifyDialog(pix, "F3-PIX");
                var qr = Children(pix).OfType<PictureBox>().Single();
                if (qr.Image == null || qr.BackColor != Color.White) throw new Exception("QR Code alterado");
                stage = 11; pix.DialogResult = DialogResult.Cancel; pix.Close(); return;
            }
            if (stage == 11) { var payment = Dialog("Finalizar Venda"); stage = 12; payment.DialogResult = DialogResult.Cancel; payment.Close(); return; }
            if (stage == 12) { stage = 13; SendKey(sales, Keys.F7); return; }
            if (stage == 13)
            {
                var remove = Dialog("Remover item da venda"); VerifyDialog(remove, "F7-remocao");
                stage = 14; Children(remove).OfType<Button>().Single(button => button.Text == "SIM, REMOVER").PerformClick(); return;
            }
            if (Children(sales).OfType<SalesItemsGrid>().Single().RowCount != 0) throw new Exception("Remocao nao funcionou");
            if (++themeIndex < themes.Length) { stage = 1; return; }
            sales.Close();
        };
        timer.Start();
        var open = typeof(MainForm).GetMethod("OpenSales", BindingFlags.Instance | BindingFlags.NonPublic)!;
        open.Invoke(owner, null);
        reopening = true; stage = 0;
        open.Invoke(owner, null);
        if (checks.Count != 43) throw new Exception("Verificacoes incompletas");
    }
}

internal static class WeatherChecks
{
    internal sealed class Handler : System.Net.Http.HttpMessageHandler
    {
        internal int GeoCount, ForecastCount;
        internal bool Fail, WrongState, Conditional;
        protected override Task<System.Net.Http.HttpResponseMessage> SendAsync(System.Net.Http.HttpRequestMessage request, CancellationToken token)
        {
            token.ThrowIfCancellationRequested();
            if (Fail) throw new System.Net.Http.HttpRequestException("offline");
            string json;
            if (request.RequestUri!.Host.Contains("nominatim"))
            {
                GeoCount++;
                bool curitiba = request.RequestUri.ToString().Contains("Curitiba");
                json = System.Text.Json.JsonSerializer.Serialize(new[] { new { lat = curitiba ? "-25.4284" : "-23.0067", lon = curitiba ? "-49.2733" : "-44.3181", address = new System.Collections.Generic.Dictionary<string,string> { ["city"] = curitiba ? "Curitiba" : "Angra dos Reis", ["ISO3166-2-lvl4"] = WrongState ? "BR-SP" : curitiba ? "BR-PR" : "BR-RJ" } } });
            }
            else
            {
                ForecastCount++; Conditional |= request.Headers.IfModifiedSince != null;
                var now = DateTimeOffset.UtcNow;
                json = System.Text.Json.JsonSerializer.Serialize(new { properties = new { meta = new { updated_at = now.ToString("O") }, timeseries = new[] { new { time = now.ToString("O"), data = new { instant = new { details = new { air_temperature = request.RequestUri.Query.Contains("-25.4284") ? 16.0 : 27.0, relative_humidity = 75.0, wind_speed = 2.5 } }, next_1_hours = new { summary = new { symbol_code = "partlycloudy_day" } } } } } } });
            }
            var result = new System.Net.Http.HttpResponseMessage(System.Net.HttpStatusCode.OK) { Content = new System.Net.Http.StringContent(json) };
            result.Content.Headers.Expires = DateTimeOffset.UtcNow.AddMinutes(15);
            result.Content.Headers.LastModified = DateTimeOffset.UtcNow.AddMinutes(-10);
            return Task.FromResult(result);
        }
    }
    internal static SalesWeatherService FixtureService() => new(new System.Net.Http.HttpClient(new Handler()), Path.Combine(Path.GetTempPath(), "weather-ui-" + Guid.NewGuid()));
    internal static async Task RunAsync()
    {
        var checks = new List<string>();
        var handler = new Handler();
        using var client = new System.Net.Http.HttpClient(handler);
        string folder = Path.Combine(Path.GetTempPath(), "weather-checks-" + Guid.NewGuid());
        var now = DateTimeOffset.UtcNow;
        var service = new SalesWeatherService(client, folder, () => now);
        var angra = await service.GetAsync("Angra dos Reis/RJ", CancellationToken.None);
        var curitiba = await service.GetAsync("Curitiba - PR", CancellationToken.None);
        if (angra.Location.State != "RJ" || curitiba.Location.State != "PR" || angra.Temperature != 27 || curitiba.Temperature != 16 || angra.Wind != 9) throw new Exception("Cidade/UF ou unidades divergentes");
        checks.Add("Cidade/UF: Angra e Curitiba sem mistura de dados OK");
        await service.GetAsync("Angra dos Reis/RJ", CancellationToken.None);
        if (handler.GeoCount != 2 || handler.ForecastCount != 2) throw new Exception("Cache nao respeitado");
        checks.Add("Cache local e prazo de atualizacao OK");
        now = now.AddMinutes(16); handler.Fail = true;
        var offline = await service.GetAsync("Angra dos Reis/RJ", CancellationToken.None);
        if (!offline.Offline || offline.CheckedAt != angra.CheckedAt || offline.Location.State != "RJ") throw new Exception("Dados offline incorretos");
        handler.Fail = false;
        var recovered = await service.GetAsync("Angra dos Reis/RJ", CancellationToken.None);
        if (recovered.Offline || !handler.Conditional || handler.GeoCount != 2) throw new Exception("Reconexao/cache de coordenadas falhou");
        checks.Add("Offline identificado e reconexao com revalidacao HTTP OK");
        bool rejected = false;
        try { SalesWeatherService.ParseCity(""); } catch (InvalidOperationException) { rejected = true; }
        if (!rejected) throw new Exception("Cidade ausente nao rejeitada");
        handler.WrongState = true;
        try { await new SalesWeatherService(client, folder + "-wrong").GetAsync("Angra dos Reis/RJ", CancellationToken.None); throw new Exception("UF incorreta aceita"); } catch (InvalidOperationException) { }
        now = now.AddDays(12); handler.Fail = true;
        try { await service.GetAsync("Angra dos Reis/RJ", CancellationToken.None); throw new Exception("Previsao vencida aceita"); } catch (InvalidOperationException) { }
        checks.Add("Cidade ausente, UF incorreta e previsao vencida rejeitadas OK");
        using var cancelled = new CancellationTokenSource(); cancelled.Cancel();
        try { await service.GetAsync("Curitiba/PR", cancelled.Token); throw new Exception("Cancelamento ignorado"); } catch (OperationCanceledException) { }
        checks.Add("Cancelamento ao fechar OK");
        var live = new SalesWeatherService(cacheFolder: folder + "-live");
        foreach (string city in new[] { "Angra dos Reis/RJ", "Curitiba/PR" })
        {
            var reading = await live.GetAsync(city, CancellationToken.None);
            checks.Add($"Online real: {reading.Location.City}/{reading.Location.State} {reading.Temperature:0}C, previsao {reading.ForecastAt:O} OK");
        }
        File.WriteAllText("visual-qa/weather-checks.json", System.Text.Json.JsonSerializer.Serialize(checks));
    }
}
